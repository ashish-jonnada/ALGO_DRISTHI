Pseudocode

for (i = 0; i < array.length - 1; i++)
swapped = false;
for (j = 0; j < array.length - 1 - i; j++)
if (array[j] > array[j + 1])
swap(array[j], array[j + 1])
swapped = true;
if (swapped == false)
break;
